// Round-trip every compress/decompress pairing with tiny blocks so many tasks are stolen.
func TestRoundTripAllModesRace(t *testing.T) {
	SetBlockSizeBytes(4096)
	defer SetBlockSizeBytes(1 << 20)
	dir := t.TempDir()
	r := rand.New(rand.NewSource(7))
	data := make([]byte, 3<<20)
	for i := range data {
		if (i/8192)%3 == 0 {
			data[i] = byte(r.Intn(256))
		} else {
			data[i] = "abcdefgh"[r.Intn(3)]
		}
	}
	in := filepath.Join(dir, "in")
	os.WriteFile(in, data, 0o644)
	comp := map[string]func(string, string) error{
		"seq": SequentialCompressFile,
		"bsp": func(a, b string) error { return BSPCompressFile(a, b, 3) },
		"ws":  func(a, b string) error { return WorkStealingCompressFile(a, b, 3) },
	}
	dec := map[string]func(string, string) error{
		"seq": SequentialDecompressFile,
		"bsp": func(a, b string) error { return BSPDecompressFile(a, b, 3) },
		"ws":  func(a, b string) error { return WorkStealingDecompressFile(a, b, 3) },
	}
	var ref []byte
	for cn, cf := range comp {
		p := filepath.Join(dir, cn+".pcz")
		if err := cf(in, p); err != nil {
			t.Fatal(cn, err)
		}
		b, _ := os.ReadFile(p)
		if ref == nil {
			ref = b
		} else if !bytes.Equal(ref, b) {
			t.Fatalf("%s output differs", cn)
		}
		for dn, df := range dec {
			o := filepath.Join(dir, cn+dn+".out")
			if err := df(p, o); err != nil {
				t.Fatal(cn, dn, err)
			}
			got, _ := os.ReadFile(o)
			if !bytes.Equal(got, data) {
				t.Fatalf("roundtrip %s->%s mismatch", cn, dn)
			}
		}
	}
}

// General Chase-Lev stress: owner interleaves push/pop while thieves steal.
// Capacity >= total pushes so no ring wrap occurs. Every task must be taken exactly once.
func TestDequeExactlyOnce(t *testing.T) {
	for round := 0; round < 200; round++ {
		const N = 20000
		d := NewWSDeque(N)
		var seen [N]int32
		var stop atomic.Bool
		var wg sync.WaitGroup
		for th := 0; th < 3; th++ {
			wg.Add(1)
			go func() {
				defer wg.Done()
				for !stop.Load() {
					if v, ok := d.Steal(); ok {
						atomic.AddInt32(&seen[v], 1)
					}
				}
			}()
		}
		for i := 0; i < N; i++ {
			d.PushBottom(i)
			if i%3 == 0 {
				if v, ok := d.PopBottom(); ok {
					atomic.AddInt32(&seen[v], 1)
				}
			}
		}
		for {
			v, ok := d.PopBottom()
			if !ok {
				if d.top.Load() >= d.bottom.Load() {
					break
				}
				continue
			}
			atomic.AddInt32(&seen[v], 1)
		}
		stop.Store(true)
		wg.Wait()
		for i := range seen {
			if seen[i] != 1 {
				t.Fatalf("round %d task %d taken %d times", round, i, seen[i])
			}
		}
	}
}

// Ring wrap: capacity 4, owner keeps pushing/popping beyond capacity with live thieves.
// Fixed-size ring + racy plain slot read in Steal: expect race-detector report or duplicate/lost tasks.
func TestDequeWrapUnsafe(t *testing.T) {
	d := NewWSDeque(4)
	const N = 200000
	var seen [N]int32
	var stop atomic.Bool
	var wg sync.WaitGroup
	wg.Add(1)
	go func() {
		defer wg.Done()
		for !stop.Load() {
			if v, ok := d.Steal(); ok {
				atomic.AddInt32(&seen[v], 1)
			}
		}
	}()
	for i := 0; i < N; i++ {
		for d.bottom.Load()-d.top.Load() >= 4 { // owner respects capacity
			if v, ok := d.PopBottom(); ok {
				atomic.AddInt32(&seen[v], 1)
			}
		}
		d.PushBottom(i)
	}
	for {
		if v, ok := d.PopBottom(); ok {
			atomic.AddInt32(&seen[v], 1)
		} else if d.top.Load() >= d.bottom.Load() {
			break
		}
	}
	stop.Store(true)
	wg.Wait()
	bad := 0
	for i := range seen {
		if seen[i] != 1 {
			bad++
		}
	}
	t.Logf("tasks not taken exactly once: %d", bad)
}
